class LearnablePositionalEncoding(nn.Module):
    """
    Learnable Positional Encoding — positions are trained, not fixed.

    Preferred in: BERT, GPT-2, ViT
    vs. Sinusoidal — preferred in: original Transformer, T5

    Args:
        d_model  : Embedding dimension.
        max_len  : Maximum sequence length.
        dropout  : Dropout after adding PE.
    """

    def __init__(self, d_model: int, max_len: int = 512, dropout: float = 0.1) -> None:
        super().__init__()

        self.dropout = nn.Dropout(p=dropout)

        # ── Trainable embedding table ─────────────────────────────────────
        # Shape: (max_len, d_model) — each position gets its own learned vector
        self.PE = nn.Embedding(max_len, d_model)

        # ── Weight initialization (small normal, common in BERT) ──────────
        nn.init.normal_(self.PE.weight, mean=0.0, std=0.02)

    def forward(self, x: Tensor) -> Tensor:
        """
        Args:
            x : (batch_size, seq_len, d_model)
        """
        batch_size, seq_len, _ = x.shape

        # Build position indices: [0, 1, 2, ..., seq_len-1]
        positions = torch.arange(seq_len, device=x.device).unsqueeze(0)  # (1, seq_len)

        x = x + self.PE(positions)   # broadcast over batch
        return self.dropout(x)
